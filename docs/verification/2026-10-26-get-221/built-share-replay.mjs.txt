import assert from 'node:assert/strict';
import fs from 'node:fs/promises';
import path from 'node:path';
import {pathToFileURL} from 'node:url';
import {createReaderServer} from '../server/server.mjs';
const artifact=(await fs.readFile('.symphony/artifact-path','utf8')).trim(),built=path.join(artifact,'dist'),out=path.join(artifact,'built-share');await fs.mkdir(out,{recursive:true});
const codes=new Map(),server=await createReaderServer({bookPath:path.join(out,'absent.json'),recordsPath:path.join(out,'records.json'),portalDataDir:path.join(out,'portal'),env:{},allowLocal:false,sendCode:async(e,c)=>codes.set(e,c),chatGenerator:null,companionGenerator:null});
const engines=await import(pathToFileURL(process.env.LEAFHOUR_PLAYWRIGHT_MODULE));const report={surface:'actual built read.html and hashed reader bundle, real local HTTP API',cases:[]};let browser;
try{await new Promise(r=>server.listen(0,'127.0.0.1',r));const base='http://127.0.0.1:'+server.address().port;for(const name of ['chromium','webkit']){
 browser=await engines[name].launch({headless:true,executablePath:process.env[name==='chromium'?'LEAFHOUR_CHROMIUM_EXECUTABLE':'LEAFHOUR_WEBKIT_EXECUTABLE']});const context=await browser.newContext({viewport:{width:390,height:900},acceptDownloads:true});
 const email='built-share-'+name+'@example.test';assert.equal((await context.request.post(base+'/api/auth/code',{headers:{Origin:base},data:{email,purpose:'login'}})).status(),200);assert.equal((await context.request.post(base+'/api/auth/verify',{headers:{Origin:base},data:{email,purpose:'login',code:codes.get(email)}})).status(),200);
 const page=await context.newPage(),item={engine:name,version:browser.version(),served:[],errors:[]};report.cases.push(item);page.on('pageerror',e=>item.errors.push(e.message));
 await page.route(base+'/**',async route=>{const request=route.request(),url=new URL(request.url());if(request.method()!=='GET'||url.pathname.startsWith('/api/'))return route.continue();const relative=url.pathname==='/read'?'read.html':url.pathname.slice(1),file=path.resolve(built,relative);assert.ok(file.startsWith(built+'/'));try{const stat=await fs.stat(file);if(!stat.isFile())return route.continue();const contentType=relative.endsWith('.html')?'text/html':/\.(js|mjs)$/.test(relative)?'text/javascript':relative.endsWith('.css')?'text/css':undefined;item.served.push(relative);await route.fulfill({status:200,body:await fs.readFile(file),...(contentType?{contentType}:{} )});}catch(error){if(error.code!=='ENOENT')throw error;await route.continue();}});
 await page.goto(base+'/read?book=gutenberg-alice&ui=en');await page.locator('#chapter-text .passage').first().waitFor();assert.ok(item.served.some(p=>/^reader-web\/reader-.*\.js$/.test(p)));assert.ok(!item.served.includes('share-model.mjs'));
 const p=page.locator('#chapter-text .passage p').first(),r=await p.boundingBox();await page.mouse.move(r.x+4,r.y+10);await page.mouse.down();await page.mouse.move(r.x+Math.min(150,r.width-8),r.y+10,{steps:5});await page.mouse.up();await page.locator('#more-actions').click();await page.locator('#show-share').click();await page.waitForFunction(()=>!document.querySelector('#share-download').disabled&&document.querySelector('#share-canvas').width>300&&document.querySelector('#share-preview-description').textContent.trim());assert.ok(item.served.includes('share-model.mjs'));const [download]=await Promise.all([page.waitForEvent('download'),page.locator('#share-download').click()]);const png=path.join(out,name+'.png');await download.saveAs(png);const bytes=await fs.readFile(png);assert.deepEqual([...bytes.subarray(0,8)],[137,80,78,71,13,10,26,10]);assert.ok(bytes.readUInt32BE(16)>300);assert.deepEqual(item.errors,[]);item.png={bytes:bytes.length,width:bytes.readUInt32BE(16),height:bytes.readUInt32BE(20)};item.passed=true;await context.close();await browser.close();browser=null;console.log(JSON.stringify({engine:name,passed:true,png:item.png}));
 }}finally{await browser?.close();server.closeAllConnections();await new Promise(r=>server.close(r));await fs.writeFile(path.join(out,'report.json'),JSON.stringify(report,null,2)+'\n');}
